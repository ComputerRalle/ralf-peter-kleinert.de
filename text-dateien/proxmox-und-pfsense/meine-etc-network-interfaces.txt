source /etc/network/interfaces.d/*

auto lo
iface lo inet loopback
iface lo inet6 loopback
iface enp6s0 inet static

# Hauptnetzwerk #####################################################################################################################
auto vmbr0
iface vmbr0 inet static
        address 148.251.42.47/27
        gateway 148.251.42.33
        bridge-ports enp6s0
        bridge-fd 0
        bridge-stp off
        post-up   echo 1 > /proc/sys/net/ipv4/ip_forward
        # Alles außer SSH (22) und GUI (8006) an pfSense weiterleiten
        post-up iptables -t nat -A PREROUTING -i vmbr0 -p tcp -m multiport ! --dport 22,8006 -j DNAT --to 10.0.0.2
        post-up iptables -t nat -A PREROUTING -i vmbr0 -p udp -j DNAT --to 10.0.0.2

        # Entfernt die Regeln beim Herunterfahren
        post-down iptables -t nat -D PREROUTING -i vmbr0 -p tcp -m multiport ! --dport 22,8006 -j DNAT --to 10.0.0.2
        post-down iptables -t nat -D PREROUTING -i vmbr0 -p udp -j DNAT --to 10.0.0.2

iface vmbr0 inet6 static
        address 2a01:4f8:201:9232::2/64
        gateway fe80::1
# Hauptnetzwerk ######################################################################################################################

auto vmbr1
iface vmbr1 inet static
        address 10.0.0.1/30
        bridge-ports none
        bridge-fd 0
        bridge-stp off
        post-up   iptables -t nat -A POSTROUTING -s '10.0.0.0/30' -o vmbr0 -j MASQUERADE
        post-down iptables -t nat -D POSTROUTING -s '10.0.0.0/30' -o vmbr0 -j MASQUERADE
# VM-Netzwerk Out

auto vmbr2
iface vmbr2 inet static
        bridge-ports none
        bridge-fd 0
        bridge-stp off
# VM-Netzwerk In
